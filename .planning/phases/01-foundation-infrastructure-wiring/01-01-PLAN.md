       ```go
       package config

       import (
           "fmt"
           "strings"

           "github.com/caarlos0/env/v11"
       )

       // Config holds all service configuration parsed from environment variables.
       // Field names and defaults match backend/src/malscan/config.py exactly.
       type Config struct {
           // Required — no defaults (security-sensitive)
           DatabaseURL   string `env:"DATABASE_URL,required"`
           MinioEndpoint string `env:"MINIO_ENDPOINT,required"`
           MinioAccessKey string `env:"MINIO_ACCESS_KEY,required"`
           MinioSecretKey string `env:"MINIO_SECRET_KEY,required"`
           RabbitmqURL   string `env:"RABBITMQ_URL,required"`

           // Optional with defaults matching Python config.py
           MinioSecure   bool   `env:"MINIO_SECURE"          envDefault:"false"`
           MinioBucket   string `env:"MINIO_BUCKET_UPLOADS"   envDefault:"uploads"`
           RabbitmqQueue string `env:"RABBITMQ_QUEUE"         envDefault:"malscan.jobs"`
           MaxFileSize   int64  `env:"MAX_FILE_SIZE"           envDefault:"104857600"` // 100MB
           CORSOrigins   string `env:"CORS_ORIGINS"            envDefault:"*"`
           LogLevel      string `env:"LOG_LEVEL"               envDefault:"INFO"`
           Port          int    `env:"PORT"                    envDefault:"8080"`
           StagesTotal   int    `env:"STAGES_TOTAL"            envDefault:"5"`
       }

       // Load parses environment variables into Config and transforms DATABASE_URL.
       // The shared .env file uses postgresql+asyncpg:// (SQLAlchemy dialect).
       // pgx requires plain postgresql:// — we strip +asyncpg here (per DB-07).
       func Load() (*Config, error) {
           cfg := &Config{}
           if err := env.Parse(cfg); err != nil {
               return nil, fmt.Errorf("parse config: %w", err)
           }

           // Strip SQLAlchemy asyncpg dialect: "postgresql+asyncpg://" → "postgresql://"
           cfg.DatabaseURL = strings.Replace(cfg.DatabaseURL, "+asyncpg", "", 1)

           return cfg, nil
       }
       ```

       - `func main()` that calls `run()` and exits with code 1 on error (logging the error via slog.Error)
       - `func run() error` that:
         a. Calls `config.Load()` — returns error on failure
         b. Calls `setupLogger(cfg.LogLevel)` to configure slog
         c. Logs `slog.Info("starting ingest service", "port", cfg.Port)`
         d. Creates a `chi.NewRouter()` with `middleware.Recoverer`
         e. Starts `http.Server` with `Addr: fmt.Sprintf(":%d", cfg.Port)`, `Handler: r`, `ReadHeaderTimeout: 10 * time.Second`
         f. Launches server in goroutine via `go func() { srv.ListenAndServe() }()`
         g. Waits for SIGINT/SIGTERM via `signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)`
         h. On signal, calls `srv.Shutdown(ctx)` with 30-second timeout
       - `func setupLogger(levelStr string)` that:
         a. Parses level string: "DEBUG"→slog.LevelDebug, "INFO"→slog.LevelInfo, "WARN"/"WARNING"→slog.LevelWarn, "ERROR"→slog.LevelError, default→slog.LevelInfo
         b. Creates `slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: level})`
         c. Creates logger with `slog.New(handler).With("service", "ingest")`
         d. Sets as default via `slog.SetDefault(logger)`
