    **Deployment resource:**

    ```yaml
    apiVersion: apps/v1
    kind: Deployment
    metadata:
      name: malscan-ingest
      namespace: malscan
    spec:
      replicas: 1                          # D-08: 1 replica for initial deployment
      selector:
        matchLabels:
          app: malscan-ingest
      template:
        metadata:
          labels:
            app: malscan-ingest
          annotations:
            prometheus.io/scrape: "true"
            prometheus.io/port: "8080"
            prometheus.io/path: "/metrics"
        spec:
          securityContext:
            runAsNonRoot: true
            runAsUser: 1000                # Matches Dockerfile adduser uid 1000
          containers:
            - name: ingest
              image: ghcr.io/daluoter/malscan-ingest:latest
              imagePullPolicy: IfNotPresent
              ports:
                - containerPort: 8080
              envFrom:                     # D-09: Reuse shared configmap + secrets
                - configMapRef:
                    name: malscan-config
                - secretRef:
                    name: malscan-secrets
              env:                         # D-09: Ingest-specific values as inline env
                - name: PORT
                  value: "8080"
                - name: MAX_FILE_SIZE
                  value: "104857600"       # 100MB
                - name: SHUTDOWN_TIMEOUT
                  value: "30s"
                - name: CORS_ORIGINS
                  value: "*"
              resources:                   # D-11: Match api resources
                requests:
                  memory: "256Mi"
                  cpu: "100m"
                limits:
                  memory: "512Mi"
                  cpu: "500m"
              startupProbe:                # Follow api pattern — up to 3 min for startup
                httpGet:
                  path: /healthz           # Ingest uses /healthz (Phase 1)
                  port: 8080
                initialDelaySeconds: 5
                periodSeconds: 10
                timeoutSeconds: 5
                failureThreshold: 18
              livenessProbe:
                httpGet:
                  path: /healthz
                  port: 8080
                periodSeconds: 10
                timeoutSeconds: 5
                failureThreshold: 3
              readinessProbe:
                httpGet:
                  path: /healthz           # Ingest has no separate /ready — use /healthz
                  port: 8080
                periodSeconds: 5
                timeoutSeconds: 5
                failureThreshold: 3
              securityContext:              # D-10: readOnly + emptyDir /tmp
                readOnlyRootFilesystem: true
                allowPrivilegeEscalation: false
              volumeMounts:
                - name: tmp-volume
                  mountPath: /tmp
          volumes:
            - name: tmp-volume
              emptyDir: {}                 # D-10: Required for upload temp file streaming
    ```

    **Service resource (after `---` separator):**

    ```yaml
    apiVersion: v1
    kind: Service
    metadata:
      name: malscan-ingest
      namespace: malscan
    spec:
      type: ClusterIP                      # D-07: Internal only, reachable via proxy
      selector:
        app: malscan-ingest
      ports:
        - port: 8080
          targetPort: 8080
    ```

    Key differences from api deployment:
    - Service type is `ClusterIP` not `NodePort` (per D-07)
    - No `nodePort` field
    - Probe paths use `/healthz` instead of `/health` and `/ready`
    - Port is 8080 not 8000
    - Ingest-specific inline env vars for PORT, MAX_FILE_SIZE, SHUTDOWN_TIMEOUT, CORS_ORIGINS
