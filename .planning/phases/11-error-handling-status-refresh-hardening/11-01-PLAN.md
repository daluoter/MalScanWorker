**4. Modify handleSubmit to track attempts and handle 429 (per D-03, D-09):**
```typescript
const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (isTerminalFailure) return  // guard: disabled form
    const password = passwordRef.current
    if (!password.trim()) return

    setIsSubmitting(true)
    setErrorMessage(null)  // clear previous error on new attempt

    try {
        const response = await apiClient.submitPassword(jobId, password)
        // Store attempt counters from response (D-03)
        setAttemptsUsed(response.attempts_used)
        setAttemptsRemaining(response.attempts_remaining)
        // Clear password immediately (SEC-04)
        passwordRef.current = ''
        if (inputRef.current) inputRef.current.value = ''
        // Notify parent to set SSE tracking flag (D-01)
        onPasswordSubmitted()
        setSuccessMessage(true)
        setTimeout(() => {
            onPasswordAccepted()
        }, 1500)
    } catch (err) {
        // Clear password on error too (SEC-04)
        passwordRef.current = ''
        if (inputRef.current) inputRef.current.value = ''
        setIsSubmitting(false)

        if (err instanceof ApiRequestError) {
            if (err.code === 'MAX_ATTEMPTS_EXCEEDED' || err.statusCode === 429) {
                // Terminal failure via 429 backup path (D-09)
                setAttemptsUsed(3)
                setAttemptsRemaining(0)
                setIsTerminalFailure(true)
                setErrorMessage('密碼嘗試次數已用完 (3/3)。請重新上傳檔案並輸入正確密碼。')
            } else {
                // Other API errors — will be detailed in Plan 02
                setErrorMessage(err.message)
            }
        } else {
            setErrorMessage('發生未知錯誤')
        }

        inputRef.current?.focus()
    }
}
```

Note: Do NOT add `finally { setIsSubmitting(false) }` anymore — on success path, `isSubmitting` stays true through the success message display. On error path, it's cleared in the catch block. On wrong-password detection (SSE cycle), it's cleared in the `wrongPasswordSignal` useEffect.

const [passwordSubmitted, setPasswordSubmitted] = useState(false)

- `passwordSubmitted`: Set to `true` when PasswordForm calls `onPasswordSubmitted` (POST returned 200). Cleared when SSE delivers a non-`password_required` status.
- `wrongPasswordSignal`: Counter incremented when SSE delivers `password_required` while `passwordSubmitted` is `true`. This is the wrong-password detection signal passed to PasswordForm.

**2. Use refs for the SSE handler to access latest state without stale closures:**

Because the `es.onmessage` handler is created once in a `useEffect` and never re-created (to avoid reconnecting EventSource), it captures stale closure values. Use refs:

```typescript
const passwordSubmittedRef = useRef(false)

// Keep ref in sync with state
const handlePasswordSubmitted = () => {
    setPasswordSubmitted(true)
    passwordSubmittedRef.current = true
}
```

**3. Modify SSE onmessage handler to detect wrong password cycle (per D-01):**
```typescript
es.onmessage = (event) => {
    try {
        const data = JSON.parse(event.data) as JobStatus
        setJob(data)

        if (data.status === 'password_required') {
            // If we just submitted a password and SSE returns password_required again,
            // the worker rejected the password (D-01, D-02)
            if (passwordSubmittedRef.current) {
                passwordSubmittedRef.current = false
                setPasswordSubmitted(false)
                setPasswordAccepted(false)  // return to form view
                setWrongPasswordSignal(prev => prev + 1)
            }
        } else {
            // Non-password status — clear tracking flags
            setPasswordSubmitted(false)
            passwordSubmittedRef.current = false
            setPasswordAccepted(false)
        }

        if (data.status === 'done') {
            es.close()
            navigate(`/reports/${jobId}`)
        } else if (data.status === 'failed') {
            es.close()
        }
    } catch {
        // ignore malformed messages
    }
}
```
