const [showSlowWarning, setShowSlowWarning] = useState(false)

**2. Add useEffect for timeout tracking (per D-13):**
```typescript
useEffect(() => {
    if (!passwordAccepted) {
        setShowSlowWarning(false)
        return
    }

    const timer = setTimeout(() => {
        // Only show warning if still at 0% progress
        // (job state is accessed via the current job ref or state)
        setShowSlowWarning(true)
    }, 30000)

    return () => clearTimeout(timer)
}, [passwordAccepted])

// Clear warning when progress moves past 0%
useEffect(() => {
    if (job && job.progress.percent > 0) {
        setShowSlowWarning(false)
    }
}, [job?.progress.percent])
```
