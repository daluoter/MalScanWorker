From backend/src/malscan/api/routes.py — Python upload flow (lines 86-276):
1. Parse form (file + optional parent_job_id)
2. Validate parent_job_id if present
3. Sanitize filename, read content_type
4. Stream chunks, hash, size check
5. Upload to MinIO keyed by SHA256
6. Dedup check: SELECT files WHERE sha256 = hash
7. If exists: reuse file_record. Else: INSERT file.
8. INSERT job (in same transaction)
9. Commit
10. Publish to RabbitMQ
11. If publish fails: mark job failed, return 503
12. Return 201 {job_id, file_id, sha256, status, created_at}

    ```go
    // 8. Parse optional parent_job_id from form
    // Read "parent_job_id" part from multipart (before finding "file" part, update the loop)
    // If present, validate: uuid.Parse, then h.store.ValidateParentJob
    // On invalid format → 400 INVALID_REQUEST "Invalid parent_job_id format"
    // On parent not found → 400 INVALID_REQUEST "Parent job not found"
    // On depth exceeded → 400 INVALID_REQUEST "Maximum recursion depth (N) reached"

    // IMPORTANT: Update the multipart loop to also capture parent_job_id text field
    // The multipart loop should read ALL parts, capturing both "file" and "parent_job_id"

    // 9. Dedup check + DB transaction
    // Before MinIO PutObject, check if file.IsNew from CreateFileAndJob
    // Actually — the Python flow uploads to MinIO FIRST, then checks DB.
    // But for dedup (STORE-03), we should check DB first and skip MinIO if duplicate.
    //
    // NEW FLOW (matches STORE-03 requirement):
    // a) Stream to temp file, compute hash (existing steps 1-6)
    // b) Call store.CreateFileAndJob — this does INSERT ON CONFLICT for files
    //    If FileRecord.IsNew == false → skip MinIO upload (dedup)
    //    If FileRecord.IsNew == true → upload to MinIO
    // c) If MinIO upload fails → we already committed the DB record...
    //
    // BETTER FLOW (matches Python more closely, safer):
    // a) Stream to temp file, compute hash (existing steps 1-6)
    // b) Upload to MinIO with SHA256 key (existing step 7) — MinIO PutObject is idempotent
    // c) Call store.CreateFileAndJob — handles dedup internally
    // d) Publish to RabbitMQ
    //
    // ACTUALLY per STORE-03: "Before uploading to MinIO, Go service checks PostgreSQL"
    // So: check DB first, skip MinIO if SHA256 exists.
    //
    // FINAL FLOW:
    // a) Stream to temp file, compute hash (steps 1-6)
    // b) CreateFileAndJob(ctx, sha256, size, filename, contentType, parentJobID, depth)
    //    → returns FileRecord (with IsNew flag) and JobRecord
    // c) If FileRecord.IsNew → upload to MinIO (step 7)
    //    If !FileRecord.IsNew → skip MinIO upload (dedup, log "file_exists")
    // d) Publish job message to RabbitMQ
    // e) If publish fails → mark job failed, return 503

    fileRec, jobRec, err := h.store.CreateFileAndJob(r.Context(), sha256Hash, fileSize,
        filename, contentType, parentJobID, depth)
    if err != nil {
        WriteError(w, http.StatusInternalServerError, CodeInternalError,
            "Failed to create records: "+err.Error(), nil)
        return
    }

    // Upload to MinIO only for new files (dedup skip for existing — STORE-03)
    if fileRec.IsNew {
        if _, err := tempFile.Seek(0, io.SeekStart); err != nil {
            WriteError(w, http.StatusInternalServerError, CodeInternalError,
                "Failed to seek temp file: "+err.Error(), nil)
            return
        }
        _, err = h.storage.PutObject(r.Context(), h.bucket, sha256Hash, tempFile, fileSize,
            minio.PutObjectOptions{ContentType: contentType})
        if err != nil {
            WriteError(w, http.StatusInternalServerError, CodeStorageError,
                "Failed to store file: "+err.Error(), nil)
            return
        }
    } else {
        h.logger.Info("file_exists", "file_id", fileRec.ID.String(), "sha256", sha256Hash)
    }

    // Publish to RabbitMQ
    err = h.publisher.Publish(r.Context(), queue.JobMessage{
        JobID:            jobRec.ID.String(),
        FileID:           fileRec.ID.String(),
        StorageKey:       sha256Hash,
        SHA256:           sha256Hash,
        OriginalFilename: filename,
    })
    if err != nil {
        // Mark job as failed in DB (MQ-04)
        _ = h.store.MarkJobFailed(r.Context(), jobRec.ID,
            "Failed to publish to queue: "+err.Error())
        WriteError(w, http.StatusServiceUnavailable, CodeQueueUnavailable,
            "Failed to submit job to processing queue. Please try again.", nil)
        return
    }

    // Success response
    h.logger.Info("file_uploaded",
        "job_id", jobRec.ID.String(),
        "file_id", fileRec.ID.String(),
        "sha256", sha256Hash,
        "size", fileSize,
        "filename", filename)

    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusOK)  // Phase 4 changes to 201
    json.NewEncoder(w).Encode(map[string]any{
        "job_id":     jobRec.ID.String(),
        "file_id":    fileRec.ID.String(),
        "sha256":     sha256Hash,
        "status":     jobRec.Status,
        "created_at": jobRec.CreatedAt.Format(time.RFC3339),
    })
    ```
