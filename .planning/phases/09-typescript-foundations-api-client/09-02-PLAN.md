    **Add `submitPassword` method to `ApiClient` class in `client.ts`:**

    ```typescript
    async submitPassword(jobId: string, password: string): Promise<PasswordSubmitResponse> {
        const url = `${this.baseUrl}/api/v1/jobs/${encodeURIComponent(jobId)}/password`

        let response: Response
        try {
            response = await fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ password }),
            })
        } catch {
            throw new ApiRequestError(
                '網路連線失敗',
                'NETWORK_ERROR',
                0,
            )
        }

        if (!response.ok) {
            let code = 'UNKNOWN_ERROR'
            let message = '密碼提交失敗'
            let details: Record<string, unknown> = {}

            try {
                const errorData = await response.json()
                const errorObj = errorData?.detail?.error || errorData?.error
                if (errorObj) {
                    code = errorObj.code || code
                    message = errorObj.message || message
                    details = errorObj.details || details
                }
            } catch {
                // Response was not JSON — use defaults
            }

            throw new ApiRequestError(message, code, response.status, details)
        }

        return response.json()
    }
    ```
